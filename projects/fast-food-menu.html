Interactive Fast-Food Menu

A complete restaurant ordering interface with category filters, live keyword search, persistent localStorage order tray, automatic tax math, and toast feedback.

Last updated: November 2026 · 10 min read · Zero dependencies

The cleanest way to build a front-end ordering app is state-driven rendering: store your menu and cart as plain JavaScript arrays, filter and compute totals in memory, and re-render the DOM whenever state changes.

What you ship: Responsive 2-column menu & order tray, multi-condition filtering (category + search), quantity stepper controls, 8% tax calculation, localStorage persistence, and accessible toast notifications.

What you will learn by building this

Tutorials often teach array methods like .filter(), .map(), and .reduce() in isolation. In a real ordering app, all three work together on every user click. Clicking a category tab or typing in the search bar runs .filter(). Rendering the matching cards runs .map(). Updating the receipt totals runs .reduce().

This project bridges the gap between basic DOM manipulation and our full four-stage POS & Billing App Build—giving you a working, portfolio-ready front-end system in a single HTML file.

  <script>
    const MENU = [
      { id: 1, name: "Classic Smash Burger", category: "burgers", price: 8.99, emoji: "🍔", desc: "Double beef patty, cheddar, house sauce" },
      { id: 2, name: "Crispy Zinger Burger", category: "burgers", price: 9.49, emoji: "🍗", desc: "Spiced chicken fillet, lettuce, spicy mayo" },
      { id: 3, name: "Loaded Cheese Fries", category: "sides", price: 4.99, emoji: "🍟", desc: "Skin-on fries with warm cheddar & jalapeños" },
      { id: 4, name: "Mozzarella Sticks", category: "sides", price: 5.49, emoji: "🧀", desc: "6 golden breaded sticks with marinara dip" },
      { id: 5, name: "Iced Mint Lemonade", category: "drinks", price: 3.49, emoji: "🥤", desc: "Fresh squeezed lemon, crushed mint, ice" },
      { id: 6, name: "Double Smash Combo", category: "combos", price: 13.99, emoji: "🔥", desc: "Smash burger + regular fries + cold drink" }
    ];

    let activeCategory = "all";
    let searchQuery = "";
    let cart = JSON.parse(localStorage.getItem("mak_food_cart")) || [];

    function showToast(msg) {
      const container = document.getElementById("toast-container");
      const toast = document.createElement("div");
      toast.className = "toast";
      toast.textContent = msg;
      container.appendChild(toast);
      void toast.offsetWidth; // Force browser reflow so CSS transition fires
      toast.classList.add("show");
      setTimeout(() => {
        toast.classList.remove("show");
        setTimeout(() => toast.remove(), 250);
      }, 2600);
    }

    function renderMenu() {
      const grid = document.getElementById("menuGrid");
      const filtered = MENU.filter(item => {
        const matchCat = activeCategory === "all" || item.category === activeCategory;
        const matchText = item.name.toLowerCase().includes(searchQuery.toLowerCase());
        return matchCat && matchText;
      });
      grid.innerHTML = filtered.map(item => `
        <article class="card">
          <div class="card-emoji">${item.emoji}</div>
          <div class="card-title">${item.name}</div>
          <p class="card-desc">${item.desc}</p>
          <div class="card-foot">
            <span class="price">$${item.price.toFixed(2)}</span>
            <button class="add-btn" onclick="addToCart(${item.id})">+ Add</button>
          </div>
        </article>
      `).join("");
    }

    function addToCart(id) {
      const existing = cart.find(c => c.id === id);
      if (existing) existing.qty += 1;
      else {
        const item = MENU.find(m => m.id === id);
        cart.push({ ...item, qty: 1 });
      }
      saveAndRenderCart();
      showToast("Added to order tray");
    }

    function updateQty(id, delta) {
      const item = cart.find(c => c.id === id);
      if (!item) return;
      item.qty += delta;
      if (item.qty <= 0) cart = cart.filter(c => c.id !== id);
      saveAndRenderCart();
    }

    function saveAndRenderCart() {
      localStorage.setItem("mak_food_cart", JSON.stringify(cart));
      const container = document.getElementById("cartItems");
      if (!cart.length) {
        container.innerHTML = `<p style="color:#94a3b8;font-size:0.85rem;">Your tray is empty.</p>`;
      } else {
        container.innerHTML = cart.map(item => `
          <div class="tray-item">
            <div><strong>${item.name}</strong><br>$${(item.price * item.qty).toFixed(2)}</div>
            <div>
              <button class="qty-btn" onclick="updateQty(${item.id}, -1)">-</button>
              <span style="margin:0 0.4rem;">${item.qty}</span>
              <button class="qty-btn" onclick="updateQty(${item.id}, 1)">+</button>
            </div>
          </div>
        `).join("");
      }
      const sub = cart.reduce((sum, item) => sum + item.price * item.qty, 0);
      const tax = sub * 0.08;
      document.getElementById("subtotal").textContent = `$${sub.toFixed(2)}`;
      document.getElementById("tax").textContent = `$${tax.toFixed(2)}`;
      document.getElementById("total").textContent = `$${(sub + tax).toFixed(2)}`;
    }

    document.getElementById("filters").addEventListener("click", e => {
      if (!e.target.dataset.cat) return;
      document.querySelectorAll(".filter-btn").forEach(b => b.classList.remove("active"));
      e.target.classList.add("active");
      activeCategory = e.target.dataset.cat;
      renderMenu();
    });

    document.getElementById("search").addEventListener("input", e => {
      searchQuery = e.target.value;
      renderMenu();
    });

    document.getElementById("clearCart").addEventListener("click", () => {
      cart = [];
      saveAndRenderCart();
      showToast("Order tray cleared");
    });

    document.getElementById("checkoutBtn").addEventListener("click", () => {
      if (!cart.length) return showToast("Add items before placing order");
      cart = [];
      saveAndRenderCart();
      showToast("Order placed! Preparing your food.");
    });

    renderMenu();
    saveAndRenderCart();
  </script>
</body>

Step-by-step build breakdown

Step 1: Two-column responsive shell

Use CSS Grid (grid-template-columns: 1.7fr 1fr) so the menu catalog gets roughly two-thirds of the desktop width while the order tray stays pinned on the right with position: sticky. At 768px and below, collapse to 1fr so mobile users scroll naturally from menu to checkout.

Step 2: Data-driven menu catalog

Never hardcode 12 food cards in HTML. Store items as an array of objects ({ id, name, category, price, emoji, desc }) and generate HTML strings with .map().join(''). Adding a new seasonal burger later means adding one line to the array instead of editing markup.

Step 3: Compound filtering (Category + Search)

Track two state variables—activeCategory and searchQuery. Inside renderMenu(), filter items where both conditions pass. That way, searching "fries" while inside the "Sides" tab narrows results accurately without resetting the active tab.

Step 4: Cart state & localStorage persistence

When a user clicks + Add, check if that id already exists in cart using .find(). Increment qty if found, or push a shallow copy with qty: 1. Save on every mutation with localStorage.setItem('mak_food_cart', JSON.stringify(cart)) so refreshing the tab never wipes an active order.

Step 5: Receipt totals with Array.reduce()

Compute the subtotal in a single expression: cart.reduce((sum, item) => sum + item.price * item.qty, 0). Multiply by 0.08 for tax, format every dollar string with .toFixed(2), and disable or guard checkout when cart.length === 0.

Step 6: Reliable toast animations with forced reflow

Appending an element and immediately adding .show often skips CSS transitions because the browser batches style calculations. Calling void toast.offsetWidth forces a synchronous layout read between insertion and class toggle, guaranteeing smooth slide-in feedback on every click.

🚀 Extension challenges

Once the base app runs locally, try adding these three upgrades before moving to full-stack builds:

Challenge 1 · Promo Code Input

Add a coupon input above the subtotal. If the user enters MAK20, apply a 20% discount before tax and show a badge confirming the savings.

Challenge 2 · Item Customization Modal

Let users pick burger doneness or drink size (Regular / Large +$1.00). Store the variant inside the cart item key so "Cola (Regular)" and "Cola (Large)" appear as separate tray lines.

⚠️ Common mistakes

  • Mutating the source MENU object directly. Always spread ({ ...item, qty: 1 }) when adding to cart so cart quantities never corrupt the catalog data.
  • Floating-point display bugs. JavaScript math like 8.99 * 3 can produce 26.970000000000002. Always format currency outputs with .toFixed(2) at render time.
  • Forgetting JSON.parse() on reload. Reading localStorage.getItem() directly returns a raw string, causing cart.find is not a function errors.
  • Toast z-index lower than sticky headers. Keep your toast container at z-index: 9999 or higher so confirmations never hide behind sticky navigation bars.

❓ Frequently asked questions

Why use Array.reduce() to calculate the order subtotal?

Array.reduce() iterates over every item in the cart array and accumulates price multiplied by quantity into a single number in one clean pass, avoiding manual loop variables.

How does localStorage keep the order tray saved after refresh?

localStorage only stores strings. By calling JSON.stringify(cart) on save and JSON.parse() on page load, the array of cart objects persists across browser reloads.

Why trigger a reflow before showing a toast notification?

When adding and removing CSS classes rapidly, browsers batch style updates. Reading offsetWidth forces a synchronous reflow so the CSS transition plays reliably every time.

How do category filtering and live search work together?

Both conditions are checked inside a single Array.filter() callback: the item must match the active category (or 'all') AND its name must include the lowercase search query.
