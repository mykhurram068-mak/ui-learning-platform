🍔 Interactive Fast-Food Menu

Build a responsive restaurant ordering interface with category tabs, live keyword search, dynamic CSS Grid cards, a persistent order tray with 8% tax calculation, and toast notifications.

Last updated: November 2026 · 7 min read

Modern ordering interfaces keep the menu and order summary visible side-by-side without page reloads. You store menu items as a JavaScript array, render filtered cards dynamically into a CSS Grid, and manage cart quantities with Array.reduce() and localStorage.

What you will build: A two-column ordering app featuring 4 category filters (All, Burgers, Sides, Drinks, Combos), live search, quantity steppers (+ / -), automatic subtotal/tax math, persistent cart storage, and instant toast confirmations.

    const MENU = [
      { id: 1, name: "Double Smash Burger", cat: "burgers", price: 8.99, emoji: "🍔", desc: "Two crispy-edged beef patties, aged cheddar, house sauce." },
      { id: 2, name: "Crispy Zinger Fillet", cat: "burgers", price: 7.49, emoji: "🍗", desc: "Spicy buttermilk chicken breast, iceberg lettuce, garlic mayo." },

      { id: 4, name: "Loaded Cheese Fries", cat: "sides", price: 4.29, emoji: "🍟", desc: "Golden skin-on fries topped with warm jalapeno cheese sauce." },
      { id: 5, name: "Crispy Onion Rings", cat: "sides", price: 3.99, emoji: "🧅", desc: "Thick-cut sweet onions in a crunchy seasoned batter." },
      { id: 6, name: "Iced Mint Lemonade", cat: "drinks", price: 2.99, emoji: "🥤", desc: "Freshly squeezed lemons blended with crushed garden mint." },
      { id: 7, name: "Chocolate Shake", cat: "drinks", price: 4.49, emoji: "🍫", desc: "Rich Belgian cocoa ice cream spun thick with whipped cream." },
      { id: 8, name: "MAK Mega Box", cat: "combos", price: 13.99, emoji: "🍱", desc: "Double Smash Burger, regular fries, 4 nuggets & chilled drink." }
    ];

    let activeCat = "all";
    let searchQuery = "";
    let cart = [];

    try {
      cart = JSON.parse(localStorage.getItem("mak_food_cart")) || [];
    } catch (e) {
      cart = [];
    }

    const menuGrid = document.getElementById("menuGrid");
    const cartItemsEl = document.getElementById("cartItems");
    const toastEl = document.getElementById("toast");
    let toastTimer;

    function showToast(msg) {
      clearTimeout(toastTimer);
      toastEl.textContent = msg;
      toastEl.classList.remove("show");
      void toastEl.offsetWidth; // Force reflow so animation replays cleanly
      toastEl.classList.add("show");
      toastTimer = setTimeout(() => toastEl.classList.remove("show"), 2600);
    }

    function renderMenu() {
      const filtered = MENU.filter(item => {
        const matchCat = activeCat === "all" || item.cat === activeCat;
        const matchSearch = item.name.toLowerCase().includes(searchQuery) ||
                            item.desc.toLowerCase().includes(searchQuery);
        return matchCat && matchSearch;
      });

      if (!filtered.length) {
        menuGrid.innerHTML = `<p style="color:#94a3b8;grid-column:1/-1;">No menu items match your filter.</p>`;
        return;
      }

      menuGrid.innerHTML = filtered.map(item => `
        <article class="food-card">
          <div>
            <div class="food-emoji">${item.emoji}</div>
            <h3 class="food-title">${item.name}</h3>
            <p class="food-desc">${item.desc}</p>
          </div>
          <div class="food-footer">
            <span class="price">$${item.price.toFixed(2)}</span>
            <button class="add-btn" onclick="addToCart(${item.id})">+ Add</button>
          </div>
        </article>
      `).join("");
    }

    function saveAndRenderCart() {
      localStorage.setItem("mak_food_cart", JSON.stringify(cart));
      const totalQty = cart.reduce((sum, item) => sum + item.qty, 0);
      const subtotal = cart.reduce((sum, item) => sum + item.price * item.qty, 0);
      const tax = subtotal * 0.08;
      const total = subtotal + tax;

      document.getElementById("cartCount").textContent = totalQty;
      document.getElementById("subtotalEl").textContent = `$${subtotal.toFixed(2)}`;
      document.getElementById("taxEl").textContent = `$${tax.toFixed(2)}`;
      document.getElementById("totalEl").textContent = `$${total.toFixed(2)}`;
      document.getElementById("checkoutBtn").disabled = cart.length === 0;

      if (!cart.length) {
        cartItemsEl.innerHTML = `<p style="color:#94a3b8;font-size:0.85rem;text-align:center;padding:1.5rem 0;">Your tray is empty.</p>`;
        return;
      }

      cartItemsEl.innerHTML = cart.map(item => `
        <div class="cart-item">
          <div>
            <div style="font-weight:700;">${item.emoji} ${item.name}</div>
            <div style="color:#94a3b8;font-size:0.78rem;">$${(item.price * item.qty).toFixed(2)}</div>
          </div>
          <div class="qty-controls">
            <button class="qty-btn" onclick="updateQty(${item.id}, -1)">-</button>
            <span>${item.qty}</span>
            <button class="qty-btn" onclick="updateQty(${item.id}, 1)">+</button>
          </div>
        </div>
      `).join("");
    }

    window.addToCart = function(id) {
      const product = MENU.find(m => m.id === id);
      const existing = cart.find(c => c.id === id);
      if (existing) {
        existing.qty += 1;
      } else {
        cart.push({ ...product, qty: 1 });
      }
      saveAndRenderCart();
      showToast(`Added ${product.name} to order`);
    };

    window.updateQty = function(id, delta) {
      const index = cart.findIndex(c => c.id === id);
      if (index === -1) return;
      cart[index].qty += delta;
      if (cart[index].qty <= 0) {
        const removed = cart.splice(index, 1)[0];
        showToast(`Removed ${removed.name}`);
      }
      saveAndRenderCart();
    };

    document.getElementById("categoryTabs").addEventListener("click", e => {
      if (!e.target.classList.contains("tab-btn")) return;
      document.querySelectorAll(".tab-btn").forEach(b => b.classList.remove("active"));
      e.target.classList.add("active");
      activeCat = e.target.dataset.cat;
      renderMenu();
    });

    document.getElementById("searchInput").addEventListener("input", e => {
      searchQuery = e.target.value.trim().toLowerCase();
      renderMenu();
    });

    document.getElementById("clearCartBtn").addEventListener("click", () => {
      if (!cart.length) return;
      cart = [];
      saveAndRenderCart();
      showToast("Order tray cleared");
    });

    document.getElementById("checkoutBtn").addEventListener("click", () => {
      const total = document.getElementById("totalEl").textContent;
      cart = [];
      saveAndRenderCart();
      showToast(`Order placed! Total charged: ${total}`);
    });

    renderMenu();
    saveAndRenderCart();
  </script>
</body>

🧱 How it works (6-step breakdown)

1. Two-column CSS Grid shell

The outer .app-container uses grid-template-columns: 1fr 360px. The left column fluidly adapts to screen width, while the right-hand order tray stays pinned at 360px with position: sticky; top: 1.5rem. Below 860px, a media query collapses it into a single stacked column for mobile screens.

2. Data-driven menu rendering

Instead of hardcoding 8 separate HTML cards, every dish lives as a plain object inside const MENU. Calling renderMenu() maps matching objects into semantic <article class="food-card"> blocks, making it effortless to add new items or connect a JSON API later.

3. Dual-state filtering (Category + Search)

Two lightweight state variables — activeCat and searchQuery — work together inside MENU.filter(). An item only renders if it satisfies both conditions, so typing "spicy" while on the "Burgers" tab narrows results accurately.

4. Cart aggregation with Array.reduce()

Every cart update runs through saveAndRenderCart(). Two .reduce() passes compute total item count and dollar subtotal in one clean loop, then derive 8% sales tax and the final total formatted with .toFixed(2).

5. Safe localStorage persistence

The cart array is serialized via JSON.stringify(cart) on every change. On startup, JSON.parse() is wrapped inside a try/catch block so private-browsing restrictions or malformed storage never crash the script.

6. Forced-reflow toast feedback

Rapidly clicking "+ Add" can skip CSS transitions if a class is removed and re-added in the same paint frame. Reading void toastEl.offsetWidth forces the browser to flush layout styles before re-applying .show, ensuring smooth replay every time.

⚠️ Common mistakes to avoid

  • Mutating original MENU objects directly. Always spread { ...product, qty: 1 } when pushing to cart so menu data stays clean.
  • Forgetting .toFixed(2) on currency math. JavaScript floating-point math produces values like 13.489999999999998 without formatting.
  • Leaving zero-quantity items in the cart array. Remove items cleanly with cart.splice(index, 1) once qty hits 0.
  • Low toast z-index. Keep #toast above sticky headers and sidebars so mobile users always see confirmation messages.

❓ Frequently asked questions

How do I filter menu items by both category and search text?

Keep two state variables (activeCategory and searchQuery) and filter your menu array using Array.prototype.filter() so an item must match both the selected category and the search string before rendering.

How do I calculate cart totals accurately in JavaScript?

Use Array.prototype.reduce() over your cart array to sum item.price * item.qty, compute tax from that subtotal, and format all currency displays with .toFixed(2) to avoid floating-point display glitches.

Why does my CSS toast animation fail to replay on repeated clicks?

Browsers batch class removals and additions within the same frame. Triggering a forced reflow with void toast.offsetWidth between removing and re-adding the active class restarts the CSS transition cleanly.

How do I persist the customer's cart across page reloads?

Serialize the cart array with JSON.stringify(cart) and store it via localStorage.setItem('mak_food_cart', ...). On page load, read and parse it inside a try/catch block so corrupted storage never breaks the app.
